   461→interface InfrastructureItem {
   462→  id: string;
   463→  name: string;
   464→  type: string;
   465→  city?: string;
   466→  coordinates: { lat: number; lng: number };
   467→  [key: string]: any;
   468→}
   469→
   470→interface MapMarkerProps {
   471→  item: InfrastructureItem;
   472→  isSelected: boolean;
   473→  onClick: () => void;
   474→}
   475→
   476→const MapMarker: React.FC<MapMarkerProps> = ({ item, isSelected, onClick }) => {
   477→  const getIcon = () => {
   478→    switch (item.type) {
   479→      case "datacenter":
   480→        return <Server className="w-4 h-4" />;
   481→      case "cable":
   482→        return <Cable className="w-4 h-4" />;
   483→      case "ixp":
   484→        return <Network className="w-4 h-4" />;
   485→      default:
   486→        return <MapPin className="w-4 h-4" />;
   487→    }
   488→  };
   489→
   490→  const getColor = () => {
   491→    switch (item.type) {
   492→      case "datacenter":
   493→        return "bg-blue-500 hover:bg-blue-600";
   494→      case "cable":
   495→        return "bg-purple-500 hover:bg-purple-600";
   496→      case "ixp":
   497→        return "bg-green-500 hover:bg-green-600";
   498→      default:
   499→        return "bg-gray-500 hover:bg-gray-600";
   500→    }
   501→  };
   502→
   503→  return (
   504→    <motion.button
   505→      initial={{ scale: 0 }}
   506→      animate={{ scale: isSelected ? 1.2 : 1 }}
   507→      whileHover={{ scale: 1.3 }}
   508→      className={`absolute transform -translate-x-1/2 -translate-y-1/2 p-2 rounded-full text-white shadow-lg ${getColor()} ${
   509→        isSelected ? "ring-2 ring-white ring-offset-2 ring-offset-gray-900" : ""
   510→      }`}
   511→      style={{
   512→        left: `${((item.coordinates.lng + 42) / 84) * 100}%`,
   513→        top: `${((item.coordinates.lat + 12) / 24) * 100}%`
   514→      }}
   515→      onClick={(e) => {
   516→        e.stopPropagation();
   517→        onClick();
   518→      }}
   519→    >
   520→      {getIcon()}
   521→    </motion.button>
   522→  );
   523→};
