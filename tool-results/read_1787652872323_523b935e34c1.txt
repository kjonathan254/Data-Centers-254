   205→function StatsPanel({ items }: { items: InfraItem[] }) {
   206→  const dcs = items.filter((i) => i.type === "datacenter");
   207→  const cables = items.filter((i) => i.type === "cable");
   208→  const ixps = items.filter((i) => i.type === "ixp");
   209→  const totalMW = dcs.reduce((s, d) => s + (d.powerMW ?? 0), 0);
   210→  const totalCapGbps = cables.reduce((s, c) => s + (c.capacityGbps ?? 0), 0);
   211→  const totalIxpPeak = ixps.reduce((s, x) => s + (x.peakGbps ?? 0), 0);
   212→  const totalMembers = ixps.reduce((s, x) => s + (x.members ?? 0), 0);
   213→
   214→  const stats = [
   215→    { label: "Data Centres", value: dcs.length.toString(), sub: `${totalMW.toFixed(1)} MW IT power` },
   216→    { label: "Cable Capacity", value: formatCapacity(totalCapGbps), sub: `${cables.length} systems` },
   217→    { label: "IXP Members", value: totalMembers.toString(), sub: `${totalIxpPeak} Gbps peak` },
   218→  ];
   219→
   220→  return (
   221→    <div className="grid grid-cols-3 gap-3 md:gap-4">
   222→      {stats.map((s) => (
   223→        <div key={s.label} className="glass-card rounded-xl p-3 md:p-4 border-cyan/10">
   224→          <p className="text-section-label mb-1">{s.label}</p>
   225→          <p className="text-foreground text-xl md:text-2xl font-bold tracking-tight">{s.value}</p>
   226→          <p className="text-muted-foreground text-xs mt-1">{s.sub}</p>
   227→        </div>
   228→      ))}
   229→    </div>
   230→  );

   425→export default function EastAfricaInfrastructureMap() {
   426→  const [country, setCountry] = useState<CountryFilter>("all");
   427→  const [typeFilter, setTypeFilter] = useState<TypeFilter>("all");
   428→  const [viewMode, setViewMode] = useState<ViewMode>("map");
   429→  const [search, setSearch] = useState("");
   430→  const [selectedCity, setSelectedCity] = useState<string | null>(null);
   431→
   432→  const filteredItems = useMemo(() => {
   433→    let result = ITEMS;
   434→    if (country !== "all") result = result.filter((i) => i.country === country);
   435→    if (typeFilter !== "all") result = result.filter((i) => i.type === typeFilter);
   436→    if (search.trim()) {
   437→      const q = search.toLowerCase();
   438→      result = result.filter((i) => i.name.toLowerCase().includes(q) || i.description.toLowerCase().includes(q));
   439→    }
   440→    return result;
   441→  }, [country, typeFilter, search]);
   442→
   443→  const cityGroups = useMemo(() => {
   444→    const map = new Map<string, InfraItem[]>();
   445→    for (const item of filteredItems) {
   446→      const arr = map.get(item.city) ?? [];
   447→      arr.push(item);
   448→      map.set(item.city, arr);
   449→    }
   450→    return map;
   451→  }, [filteredItems]);
   452→
   453→  const visibleCityIds = useMemo(
   454→    () => new Set(cityGroups.keys()),
   455→    [cityGroups],
   456→  );
   457→
   458→  const selectedCityItems = useMemo(
   459→    () => (selectedCity ? cityGroups.get(selectedCity) ?? [] : []),
   460→    [selectedCity, cityGroups],
   461→  );
   462→
   463→  const countryPaths = useMemo(
   464→    () =>
   465→      (Object.entries(COUNTRIES) as [CountryKey, [number, number][]][]).map(
   466→        ([key, coords]) => ({ key, d: toPath(coords) }),
   467→      ),
   468→    [],
   469→  );
   470→
   471→  const gridLines = useMemo(() => {
   472→    const lines: { x1: number; y1: number; x2: number; y2: number }[] = [];
   473→    for (let lat = -12; lat <= 15; lat += 3) {
   474→      const y = proj(lat, 28).y;
   475→      lines.push({ x1: 0, y1: y, x2: S, y2: y });
   476→    }
   477→    for (let lng = 28; lng <= 43; lng += 3) {
   478→      const x = proj(15, lng).x;
   479→      lines.push({ x1: x, y1: 0, x2: x, y2: S });
   480→    }
   481→    return lines;
   482→  }, []);
   483→
   484→  const routeData = useMemo(() => {
   485→    const countriesToShow = country === "all" ? ALL_COUNTRIES : [country];
   486→    const countrySet = new Set(countriesToShow);
   487→    return CABLE_ROUTES.filter((r) => {
   488→      const from = CITIES[r.fromCity];
   489→      const to = CITIES[r.toCity];
   490→      if (from && !countrySet.has(from.country)) return false;
   491→      if (to && !countrySet.has(to.country)) return false;
   492→      if (r.fromCity === "djibouti" && country !== "all" && country !== "ethiopia") return false;
   493→      return true;
   494→    }).map((r) => {
   495→      const p1 = proj(r.fromLat, r.fromLng);
   496→      const p2 = proj(r.toLat, r.toLng);
   497→      return { ...r, x1: p1.x, y1: p1.y, x2: p2.x, y2: p2.y };
   498→    });
   499→  }, [country]);
   500→
   501→  const handleMarkerClick = useCallback((cityId: string) => {
   502→    setSelectedCity((prev) => (prev === cityId ? null : cityId));
   503→  }, []);
   504→
   505→  return (
   506→    <section className="section-y">
   507→      <div className="max-w-7xl mx-auto px-4 md:px-6">
   508→        {/* Header */}
   509→        <div className="mb-8">
   510→          <p className="text-section-label mb-3">Infrastructure</p>
   511→          <h2 className="text-display-sm text-foreground mb-3">East Africa Infrastructure Map</h2>
   512→          <p className="text-subtitle-center">Data centres, submarine cables, and internet exchange points across Kenya, Tanzania, Uganda, Rwanda, and Ethiopia.</p>
   513→        </div>
   514→
   515→        {/* Stats */}
   516→        <div className="mb-6">
   517→          <StatsPanel items={filteredItems} />
   518→        </div>
   519→
   520→        {/* Filters */}
   521→        <div className="mb-6">
   522→          <FilterBar
   523→            country={country} setCountry={setCountry}
   524→            typeFilter={typeFilter} setTypeFilter={setTypeFilter}
   525→            viewMode={viewMode} setViewMode={setViewMode}
   526→            search={search} setSearch={setSearch}
   527→          />
   528→        </div>
   529→
   530→        {/* Content */}
   531→        {viewMode === "map" ? (
   532→          <div className="relative">
   533→            <div className="relative bg-[#0a0a0f] rounded-xl border border-border/30 overflow-hidden min-h-[400px] md:min-h-[600px]">
   534→              <svg viewBox="0 0 1000 1000" className="w-full h-full absolute inset-0" preserveAspectRatio="xMidYMid meet">
   535→                {/* Grid */}
   536→                {gridLines.map((l, i) => (
   537→                  <line key={i} x1={l.x1} y1={l.y1} x2={l.x2} y2={l.y2}
   538→                    stroke="rgba(255,255,255,0.03)" strokeWidth={0.5} />
   539→                ))}
   540→                {/* Countries */}
   541→                {countryPaths.map(({ key, d }) => {
   542→                  const show = country === "all" || country === key;
   543→                  return (
   544→                    <g key={key} opacity={show ? 1 : 0.15}>
   545→                      <path d={d} fill="rgba(6,182,212,0.04)" stroke="rgba(6,182,212,0.12)" strokeWidth={1} />
   546→                    </g>
   547→                  );
   548→                })}
   549→                {/* Cable routes */}
   550→                {routeData.map((r, i) => (
   551→                  <line key={`route-${i}`} x1={r.x1} y1={r.y1} x2={r.x2} y2={r.y2}
   552→                    stroke={r.type === "submarine" ? "rgba(168,85,247,0.25)" : "rgba(168,85,247,0.15)"}
   553→                    strokeWidth={1.5} strokeDasharray={r.type === "submarine" ? "8,6" : "6,4"} />
   554→                ))}
   555→                {/* City markers */}
   556→                {Object.entries(CITIES).map(([cityId, city]) => {
   557→                  if (!visibleCityIds.has(cityId)) return null;
   558→                  const { x, y } = proj(city.lat, city.lng);
   559→                  const items = cityGroups.get(cityId) ?? [];
   560→                  const isSelected = selectedCity === cityId;
   561→                  const hasDc = items.some((i) => i.type === "datacenter");
   562→                  const hasCable = items.some((i) => i.type === "cable");
   563→                  const hasIxp = items.some((i) => i.type === "ixp");
   564→                  return (
   565→                    <g key={cityId} className="cursor-pointer" onClick={() => handleMarkerClick(cityId)}>
   566→                      {/* Pulse ring for selected */}
   567→                      {isSelected && (
   568→                        <circle cx={x} cy={y} r={12} fill="none" stroke="rgba(6,182,212,0.3)" strokeWidth={1.5}>
   569→                          <animate attributeName="r" from="8" to="20" dur="1.5s" repeatCount="indefinite" />
   570→                          <animate attributeName="opacity" from="0.6" to="0" dur="1.5s" repeatCount="indefinite" />
   571→                        </circle>
   572→                      )}
   573→                      {/* Type indicators - stacked vertically */}
   574→                      {hasDc && (
   575→                        <circle cx={x} cy={y - 8} r={4.5} fill="rgb(6,182,212)" stroke="rgba(6,182,212,0.4)" strokeWidth={1} />
   576→                      )}
   577→                      {hasCable && (
   578→                        <circle cx={x - (hasDc && hasIxp ? 6 : hasDc || hasIxp ? 6 : 0)} cy={y + (hasDc || hasIxp ? 2 : 0)} r={3.5} fill="rgb(168,85,247)" stroke="rgba(168,85,247,0.4)" strokeWidth={1} />
   579→                      )}
   580→                      {hasIxp && (
   581→                        <circle cx={x + (hasDc && hasCable ? 6 : hasDc || hasCable ? 6 : 0)} cy={y + (hasDc || hasCable ? 2 : 0)} r={3.5} fill="rgb(16,185,129)" stroke="rgba(16,185,129,0.4)" strokeWidth={1} />
   582→                      )}
   583→                      {/* Count badge */}
   584→                      <circle cx={x + 10} cy={y - 10} r={7} fill="#0a0a0f" stroke="rgba(6,182,212,0.3)" strokeWidth={0.8} />
   585→                      <text x={x + 10} y={y - 10} textAnchor="middle" dominantBaseline="central"
   586→                        fill="oklch(0.93 0.01 260)" fontSize={8} fontWeight={600}>
   587→                        {items.length}
   588→                      </text>
   589→                      {/* City label */}
   590→                      <text x={x} y={y + 18} textAnchor="middle" fill="rgba(255,255,255,0.5)" fontSize={9} fontWeight={500}>
   591→                        {city.name}
   592→                      </text>
   593→                    </g>
   594→                  );
   595→                })}
   596→                {/* Djibouti label for Ethiopia route */
   597→                {routeData.some((r) => r.fromCity === "djibouti") && {
   598→                  const p = proj(11.5, 42.0);
   599→                  return (
   600→                    <g key="djibouti">
   601→                      <circle cx={p.x} cy={p.y} r={3} fill="rgb(168,85,247)" opacity={0.5} />
   602→                      <text x={p.x} y={p.y + 12} textAnchor="middle" fill="rgba(255,255,255,0.35)" fontSize={8}>Djibouti</text>
   603→                    </g>
   604→                  );
   605→                }}
   606→              </svg>
   607→              {/* Detail Panel overlay */}
   608→              <AnimatePresence>
   609→                {selectedCity && selectedCityItems.length > 0 && (
   610→                  <DetailPanel city={selectedCity} items={selectedCityItems} onClose={() => setSelectedCity(null)} />
   611→                )}
   612→              </AnimatePresence>
   613→            </div>
   614→            {/* Legend below map */}
   615→            <div className="mt-3 flex items-center justify-between">
   616→              <Legend />
   617→              <p className="text-[10px] text-muted-foreground hidden sm:block">
   618→                Simplified borders · Projection: equirectangular
   619→              </p>
   620→            </div>
   621→          </div>
   622→        ) : (
   623→          <ListView items={filteredItems} />
   624→        )}
   625→      </div>
   626→    </section>
