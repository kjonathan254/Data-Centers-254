    24→// ─── calcTitleLayout (from design-system.md) ───
    25→function splitTitleLines(title, charsPerLine) {
    26→  if (title.length <= charsPerLine) return [title];
    27→  const breakAfter = new Set([
    28→    ...' \t', '-', '_', '/', '(', ')', ':', ',', '.', ';',
    29→  ]);
    30→  const lines = [];
    31→  let remaining = title;
    32→  while (remaining.length > charsPerLine) {
    33→    let breakAt = -1;
    34→    for (let i = charsPerLine; i >= Math.floor(charsPerLine * 0.6); i--) {
    35→      if (i < remaining.length && breakAfter.has(remaining[i - 1])) { breakAt = i; break; }
    36→    }
    37→    if (breakAt === -1) {
    38→      const limit = Math.min(remaining.length, Math.ceil(charsPerLine * 1.3));
    39→      for (let i = charsPerLine + 1; i < limit; i++) {
    40→        if (breakAfter.has(remaining[i - 1])) { breakAt = i; break; }
    41→      }
    42→    }
    43→    if (breakAt === -1) breakAt = charsPerLine;
    44→    lines.push(remaining.slice(0, breakAt).trim());
    45→    remaining = remaining.slice(breakAt).trim();
    46→  }
    47→  if (remaining) lines.push(remaining);
    48→  if (lines.length > 1 && lines[lines.length - 1].length <= 3) {
    49→    const last = lines.pop(); lines[lines.length - 1] += " " + last;
    50→  }
    51→  return lines;
    52→}

    54→function calcTitleLayout(title, maxWidthTwips, preferredPt = 38, minPt = 24) {
    55→  const charWidth = (pt) => pt * 11; // English chars are ~half CJK width
    56→  const charsPerLine = (pt) => Math.floor(maxWidthTwips / charWidth(pt));
    57→  let titlePt = preferredPt;
    58→  let lines;
    59→  while (titlePt >= minPt) {
    60→    const cpl = charsPerLine(titlePt);
    61→    if (cpl < 2) { titlePt -= 2; continue; }
    62→    lines = splitTitleLines(title, cpl);
    63→    if (lines.length <= 3) break;
    64→    titlePt -= 2;
    65→  }
    66→  if (!lines || lines.length > 3) {
    67→    lines = splitTitleLines(title, charsPerLine(minPt));
    68→    titlePt = minPt;
    69→  }
    70→  return { titlePt, titleLines: lines };
    71→}

    73→function calcCoverSpacing(params) {
    74→  const { titleLineCount = 1, titlePt = 36, hasSubtitle = false,
    75→    hasEnglishLabel = false, metaLineCount = 0, fixedHeight = 800,
    76→    marginTop = 0, marginBottom = 0, pageHeight = 16838 } = params;
    77→  const SAFETY = 1200;
    78→  const usableHeight = pageHeight - marginTop - marginBottom - SAFETY;
    79→  const titleHeight = titleLineCount * (titlePt * 23 + 200);
    80→  const subtitleHeight = hasSubtitle ? (12 * 23 + 600) : 0;
    81→  const englishLabelHeight = hasEnglishLabel ? (9 * 23 + 600) : 0;
    82→  const metaHeight = metaLineCount * (10 * 23 + 100);
    83→  const implicitParaHeight = 3 * 300;
    84→  const contentHeight = titleHeight + subtitleHeight + englishLabelHeight + metaHeight + fixedHeight + implicitParaHeight;
    85→  const remainingSpace = usableHeight - contentHeight;
    86→  const safeRemaining = Math.max(remainingSpace, 400);
    87→  const FOOTER_MIN = 800;
    88→  const rawTop = Math.floor(safeRemaining * 0.45);
    89→  const rawBottom = Math.floor(safeRemaining * 0.45);
    90→  const bottomSpacing = Math.max(rawBottom, FOOTER_MIN);
    91→  const topSpacing = Math.max(rawTop - Math.max(0, FOOTER_MIN - rawBottom), 400);
    92→  const midSpacing = Math.max(safeRemaining - topSpacing - bottomSpacing, 0);
    93→  return { topSpacing, midSpacing, bottomSpacing };
    94→}
