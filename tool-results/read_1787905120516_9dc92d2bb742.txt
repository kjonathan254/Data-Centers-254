   212→function StatsPanel({ items }: { items: InfraItem[] }) {
   213→  const dcs = items.filter((i) => i.type === "datacenter");
   214→  const cables = items.filter((i) => i.type === "cable");
   215→  const ixps = items.filter((i) => i.type === "ixp");
   216→  const totalMW = dcs.reduce((s, d) => s + (d.powerMW ?? 0), 0);
   217→  const totalCapGbps = cables.reduce((s, c) => s + (c.capacityGbps ?? 0), 0);
   218→  const totalIxpPeak = ixps.reduce((s, x) => s + (x.peakGbps ?? 0), 0);
   219→  const totalMembers = ixps.reduce((s, x) => s + (x.members ?? 0), 0);
   220→
   221→  const stats = [
   222→    { label: "Data Centres", value: dcs.length.toString(), sub: `${totalMW.toFixed(1)} MW IT power` },
   223→    { label: "Cable Capacity", value: formatCapacity(totalCapGbps), sub: `${cables.length} systems` },
   224→    { label: "IXP Members", value: totalMembers.toString(), sub: `${totalIxpPeak} Gbps peak` },
   225→  ];
   226→
   227→  return (
   228→    <div className="grid grid-cols-3 gap-3 md:gap-4">
   229→      {stats.map((s) => (
   230→        <div key={s.label} className="glass-card rounded-xl p-3 md:p-4 border-cyan/10">
   231→          <p className="text-section-label mb-1">{s.label}</p>

   437→export default function EastAfricaInfrastructureMap() {

   440→  const [viewMode, setViewMode] = useState<ViewMode>("map");

   443→
   444→  const filteredItems = useMemo(() => {
   445→    let result = ITEMS;
   446→    if (country !== "all") result = result.filter((i) => i.country === country);
   447→    if (typeFilter !== "all") result = result.filter((i) => i.type === typeFilter);
   448→    if (search.trim()) {
   449→      const q = search.toLowerCase();
   450→      result = result.filter((i) => i.name.toLowerCase().includes(q) || i.description.toLowerCase().includes(q));
   451→    }
   452→    return result;
   453→  }, [country, typeFilter, search]);
   454→
   455→  const cityGroups = useMemo(() => {
   456→    const map = new Map<string, InfraItem[]>();
   457→    for (const item of filteredItems) {
   458→      const arr = map.get(item.city) ?? [];
   459→      arr.push(item);
   460→      map.set(item.city, arr);
   461→    }
   462→    return map;
   463→  }, [filteredItems]);
   464→
   465→  const visibleCityIds = useMemo(
   466→    () => new Set(cityGroups.keys()),
   467→    [cityGroups],
   468→  );
   469→
   470→  const selectedCityItems = useMemo(
   471→    () => (selectedCity ? cityGroups.get(selectedCity) ?? [] : []),
   472→    [selectedCity, cityGroups],
   473→  );
   474→
   475→  const countryPaths = useMemo(
   476→    () =>
   477→      (Object.entries(COUNTRIES) as [CountryKey, [number, number][]][]).map(
   478→        ([key, coords]) => ({ key, d: toPath(coords) }),
   479→      ),
   480→    [],
   481→  );
   482→
   483→  const gridLines = useMemo(() => {
   484→    const lines: { x1: number; y1: number; x2: number; y2: number }[] = [];
   485→    for (let lat = -12; lat <= 15; lat += 3) {
   486→      const y = proj(lat, 28).y;
   487→      lines.push({ x1: 0, y1: y, x2: S, y2: y });
   488→    }
   489→    for (let lng = 28; lng <= 43; lng += 3) {
   490→      const x = proj(15, lng).x;
   491→      lines.push({ x1: x, y1: 0, x2: x, y2: S });
   492→    }
   493→    return lines;
   494→  }, []);
   495→
   496→  const routeData = useMemo(() => {
   497→    const countriesToShow = country === "all" ? ALL_COUNTRIES : [country];
   498→    const countrySet = new Set(countriesToShow);
   499→    return CABLE_ROUTES.filter((r) => {
   500→      const from = CITIES[r.fromCity];
   501→      const to = CITIES[r.toCity];
   502→      if (from && !countrySet.has(from.country)) return false;
   503→      if (to && !countrySet.has(to.country)) return false;
   504→      if (r.fromCity === "djibouti" && country !== "all" && country !== "ethiopia") return false;
   505→      return true;

   509→      return { ...r, d, points: pts };
   510→    });
   511→  }, [country]);
   512→
   513→  const handleMarkerClick = useCallback((cityId: string) => {
   514→    setSelectedCity((prev) => (prev === cityId ? null : cityId));
   515→  }, []);
   516→
   517→  return (
   518→    <section className="section-y">
   519→      <div className="max-w-7xl mx-auto px-4 md:px-6">
   520→        {/* Header */}
   521→        <div className="mb-8">
   522→          <p className="text-section-label mb-3">Infrastructure</p>
   523→          <h2 className="text-display-sm text-foreground mb-3">East Africa Infrastructure Map</h2>
   524→          <p className="text-subtitle-center">Data centres, submarine cables, and internet exchange points across Kenya, Tanzania, Uganda, Rwanda, and Ethiopia.</p>
   525→        </div>
   526→
   527→        {/* Stats */}
   528→        <div className="mb-6">
   529→          <StatsPanel items={filteredItems} />
   530→        </div>
   531→
   532→        {/* Filters */}
   533→        <div className="mb-6">
   534→          <FilterBar
   535→            country={country} setCountry={setCountry}
   536→            typeFilter={typeFilter} setTypeFilter={setTypeFilter}
   537→            viewMode={viewMode} setViewMode={setViewMode}
   538→            search={search} setSearch={setSearch}
   539→          />
   540→        </div>
   541→
   542→        {/* Content */}
   543→        {viewMode === "map" ? (

   556→                    <g key={key} opacity={show ? 1 : 0.15}>

   564→                    <path d={r.d} fill="none"

   642→      </div>
